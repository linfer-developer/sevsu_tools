WITH 
    {НАЗВАНИЕ_ТАБЛИЦЫ}_ids AS (
        WITH {НАЗВАНИЕ_ТАБЛИЦЫ}({СТОЛБЦЫ}) AS (
            VALUES {ЗНАЧЕНИЯ}
        )
        INSERT INTO {НАЗВАНИЕ_ТАБЛИЦЫ_КУДА_ВСТАВКА} ({СТОЛБЦЫ})
        SELECT {ids.СТОЛБЦЫ}
        FROM {НАЗВАНИЕ_ТАБЛИЦЫ}_ids ids
        ON CONFLICT ({УНИКАЛЬНЫЕ ПОЛЯ}) DO UPDATE SET
            ФИКТИВНОЕ_ОБНОВЛЕНИЕ_ЛЮБОЕ_ПОЛЕ = EXCLUDED.ФИКТИВНОЕ_ОБНОВЛЕНИЕ_ЛЮБОЕ_ПОЛЕ
        RETURNING id, КЛЮЧ
    ),

    ...

    {НАЗВАНИЕ_ТАБЛИЦЫ}_ids AS (
        INSERT INTO {НАЗВАНИЕ_ТАБЛИЦЫ} ({СТОЛБЦЫ})
        VALUES {((SELECT id FROM {НАЗВАНИЕ_ТАБЛИЦЫ}_ids WHERE {НАЗВАНИЕ_ТАБЛИЦЫ}_key = {КЛЮЧ}), ЗНАЧЕНИЯ...)}
        ON CONFLICT ({УНИКАЛЬНЫЕ ПОЛЯ}) DO UPDATE SET
            ФИКТИВНОЕ_ОБНОВЛЕНИЕ_ЛЮБОЕ_ПОЛЕ = EXCLUDED.ФИКТИВНОЕ_ОБНОВЛЕНИЕ_ЛЮБОЕ_ПОЛЕ
        RETURNING id, КЛЮЧ
    ),

    ...
